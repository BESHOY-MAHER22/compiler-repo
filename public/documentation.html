Section 1 The Regular Expressions for Tokens Only

The lexical specification of the Rynix programming language defines all valid token classes accepted by the compiler front-end. In accordance with formal language theory, each token class is defined over the input alphabet $\Sigma$ using Regular Expressions (RE).

1.1 Auxiliary Definitions and Character Classes

Definition Name Regular Expression Description
digit [0-9] Any decimal digit from 0 through 9
nonzero_digit [1-9] Non-zero decimal digit from 1 through 9
letter [a-zA-Z] Any uppercase or lowercase English letter
letter_or_underscore [a-zA-Z_] Permissible initial character for identifiers
alphanumeric [a-zA-Z0-9_] Body character for identifiers
whitespace [ \t\r\n]+ Ignored tokens separating lexemes

1.2 Token Specification Table

The table below lists the formal regular expressions for every distinct token category recognized by the Rynix lexical analyzer:

StringLiteral " ([^"\\\n\r] | \\.)* " "Hello Rynix", "Result:\n"

MultiLineComment /\* ( [^*] | \*+ [^*/] )* \*+/ /* block comment */

Standard I/O print, input
Built-in Data Types Int, Float, String, Char, Bool, Void, Array
Special Constants true, false, null

Section 2 Deterministic Finite Automata (DFAs) of Tokens Only

A Deterministic Finite Automaton (DFA) is formally denoted as a 5-tuple $M = (Q, \Sigma, \delta, q_0, F)$, where: $Q$ is a finite set of states, $\Sigma$ is the input character set, $\delta: Q \times \Sigma \to Q$ is the state transition function, $q_0 \in Q$ is the initial state, and $F \subseteq Q$ is the set of accepting (final) states. Below are the exact vector transition diagrams for all token families in Rynix.

DFA 1: Identifier and Reserved Keyword Recognizer
States: $q_0$ (Start), $q_1$ (Accepting Identifier/Keyword). If recognized lexeme matches a keyword table entry, it is emitted as its specific Token Class; otherwise, as Identifier.
DFA 2: Numeric Constants (Integers, Decimals & Range Check)
Distinguishes IntConstant (e.g. 12), FloatConstant (e.g. 15.75), and the range operator .. (e.g. 1..5).
DFA 3: String Literal Recognizer (with Escape Sequences)
Handles double-quoted strings, escape characters (\", \\, \n), and reports unclosed string errors.
DFA 4: Comments Recognizer (Single-Line & Multi-Line)
Handles single-line comments // ... \n and delimited multi-line comments /* ... */.
DFA 5: Relational & Logical Operators (Maximal Munch / Lookahead)
Resolves ambiguous prefixes: = vs ==, ! vs !=, < vs <=, > vs >=, & vs &&, | vs ||.

3.1 C# Implementation (Rynix Scanner for .NET / Visual Studio)

This code replaces and completes the C# template under JASON_Compiler/Scanner.cs: 

using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;

namespace Rynix_Compiler
{
    public enum Token_Class
    {
        // Keywords
        Var, Fun, Class, Init, New, If, Else, While, For, In,
        Return, Break, Continue, Print, Input,
        TypeInt, TypeFloat, TypeString, TypeChar, TypeBool, TypeVoid, TypeArray,
        BoolLiteral, NullLiteral,

        // Literals & Identifiers
        Identifier, IntConstant, FloatConstant, StringLiteral, CharLiteral,

        // Operators
        PlusOp, MinusOp, MultiplyOp, DivideOp, ModuloOp,
        EqualEqualOp, NotEqualOp, LessThanOp, LessOrEqualOp, GreaterThanOp, GreaterOrEqualOp,
        AndOp, OrOp, NotOp, AssignmentOp,

        // Delimiters
        LParanthesis, RParanthesis, LBrace, RBrace, LBracket, RBracket,
        Colon, Semicolon, Comma, Dot, DotDot,

        // Comments & Error
        SingleLineComment, MultiLineComment, Unknown
    }

    public class Token
    {
        public int Id { get; set; }
        public string Lexeme { get; set; }
        public Token_Class TokenType { get; set; }
        public int Line { get; set; }
        public int Column { get; set; }
    }

    public class CompilerError
    {
        public int Line { get; set; }
        public int Column { get; set; }
        public string Lexeme { get; set; }
        public string Message { get; set; }
    }

    public class RynixScanner
    {
        public List<Token> Tokens = new List<Token>();
        public List<CompilerError> Errors = new List<CompilerError>();
        private Dictionary<string, Token_Class> Keywords = new Dictionary<string, Token_Class>();

        public RynixScanner()
        {
            // Initialize 24 reserved keywords
            Keywords.Add("var", Token_Class.Var);
            Keywords.Add("fun", Token_Class.Fun);
            Keywords.Add("class", Token_Class.Class);
            Keywords.Add("init", Token_Class.Init);
            Keywords.Add("new", Token_Class.New);
            Keywords.Add("if", Token_Class.If);
            Keywords.Add("else", Token_Class.Else);
            Keywords.Add("while", Token_Class.While);
            Keywords.Add("for", Token_Class.For);
            Keywords.Add("in", Token_Class.In);
            Keywords.Add("return", Token_Class.Return);
            Keywords.Add("break", Token_Class.Break);
            Keywords.Add("continue", Token_Class.Continue);
            Keywords.Add("print", Token_Class.Print);
            Keywords.Add("input", Token_Class.Input);
            Keywords.Add("Int", Token_Class.TypeInt);
            Keywords.Add("Float", Token_Class.TypeFloat);
            Keywords.Add("String", Token_Class.TypeString);
            Keywords.Add("Char", Token_Class.TypeChar);
            Keywords.Add("Bool", Token_Class.TypeBool);
            Keywords.Add("Void", Token_Class.TypeVoid);
            Keywords.Add("Array", Token_Class.TypeArray);
            Keywords.Add("true", Token_Class.BoolLiteral);
            Keywords.Add("false", Token_Class.BoolLiteral);
            Keywords.Add("null", Token_Class.NullLiteral);
        }

        public void StartScanning(string sourceCode)
        {
            Tokens.Clear();
            Errors.Clear();

            int i = 0;
            int line = 1;
            int col = 1;
            int tokenId = 1;
            int length = sourceCode.Length;

            while (i < length)
            {
                char ch = sourceCode[i];

                if (ch == '\n') { line++; col = 1; i++; continue; }
                if (ch == '\r') { i++; continue; }
                if (ch == ' ' || ch == '\t') { col++; i++; continue; }

                // Comments: // or /* ... */
                if (ch == '/' && i + 1 < length)
                {
                    if (sourceCode[i + 1] == '/')
                    {
                        string cText = "//";
                        int sCol = col;
                        i += 2; col += 2;
                        while (i < length && sourceCode[i] != '\n' && sourceCode[i] != '\r')
                        {
                            cText += sourceCode[i];
                            i++; col++;
                        }
                        Tokens.Add(new Token { Id = tokenId++, Lexeme = cText, TokenType = Token_Class.SingleLineComment, Line = line, Column = sCol });
                        continue;
                    }
                    if (sourceCode[i + 1] == '*')
                    {
                        string cText = "/*";
                        int sCol = col; int sLine = line;
                        i += 2; col += 2;
                        bool closed = false;
                        while (i < length)
                        {
                            if (sourceCode[i] == '*' && i + 1 < length && sourceCode[i + 1] == '/')
                            {
                                cText += "*/";
                                i += 2; col += 2;
                                closed = true;
                                break;
                            }
                            if (sourceCode[i] == '\n') { line++; col = 1; } else { col++; }
                            cText += sourceCode[i];
                            i++;
                        }
                        if (!closed)
                            Errors.Add(new CompilerError { Line = sLine, Column = sCol, Lexeme = cText, Message = "Unterminated comment block" });
                        else
                            Tokens.Add(new Token { Id = tokenId++, Lexeme = cText, TokenType = Token_Class.MultiLineComment, Line = sLine, Column = sCol });
                        continue;
                    }
                }

                // String Literals
                if (ch == '"')
                {
                    int sLine = line; int sCol = col;
                    string strVal = "";
                    i++; col++;
                    bool closed = false;
                    while (i < length)
                    {
                        if (sourceCode[i] == '"') { i++; col++; closed = true; break; }
                        if (sourceCode[i] == '\n') break;
                        if (sourceCode[i] == '\\' && i + 1 < length)
                        {
                            strVal += sourceCode[i].ToString() + sourceCode[i + 1].ToString();
                            i += 2; col += 2;
                            continue;
                        }
                        strVal += sourceCode[i];
                        i++; col++;
                    }
                    if (!closed)
                        Errors.Add(new CompilerError { Line = sLine, Column = sCol, Lexeme = "\"" + strVal, Message = "Unterminated string literal" });
                    else
                        Tokens.Add(new Token { Id = tokenId++, Lexeme = "\"" + strVal + "\"", TokenType = Token_Class.StringLiteral, Line = sLine, Column = sCol });
                    continue;
                }

                // Range operator '..'
                if (ch == '.' && i + 1 < length && sourceCode[i + 1] == '.')
                {
                    Tokens.Add(new Token { Id = tokenId++, Lexeme = "..", TokenType = Token_Class.DotDot, Line = line, Column = col });
                    i += 2; col += 2; continue;
                }

                // Two-character operators
                if (i + 1 < length)
                {
                    string two = sourceCode.Substring(i, 2);
                    Token_Class? tc = null;
                    if (two == "==") tc = Token_Class.EqualEqualOp;
                    else if (two == "!=") tc = Token_Class.NotEqualOp;
                    else if (two == "<=") tc = Token_Class.LessOrEqualOp;
                    else if (two == ">=") tc = Token_Class.GreaterOrEqualOp;
                    else if (two == "&&") tc = Token_Class.AndOp;
                    else if (two == "||") tc = Token_Class.OrOp;

                    if (tc.HasValue)
                    {
                        Tokens.Add(new Token { Id = tokenId++, Lexeme = two, TokenType = tc.Value, Line = line, Column = col });
                        i += 2; col += 2; continue;
                    }
                }

                // Single-character delimiters and operators
                switch (ch)
                {
                    case '+': Tokens.Add(new Token { Id = tokenId++, Lexeme = "+", TokenType = Token_Class.PlusOp, Line = line, Column = col }); i++; col++; continue;
                    case '-': Tokens.Add(new Token { Id = tokenId++, Lexeme = "-", TokenType = Token_Class.MinusOp, Line = line, Column = col }); i++; col++; continue;
                    case '*': Tokens.Add(new Token { Id = tokenId++, Lexeme = "*", TokenType = Token_Class.MultiplyOp, Line = line, Column = col }); i++; col++; continue;
                    case '/': Tokens.Add(new Token { Id = tokenId++, Lexeme = "/", TokenType = Token_Class.DivideOp, Line = line, Column = col }); i++; col++; continue;
                    case '%': Tokens.Add(new Token { Id = tokenId++, Lexeme = "%", TokenType = Token_Class.ModuloOp, Line = line, Column = col }); i++; col++; continue;
                    case '=': Tokens.Add(new Token { Id = tokenId++, Lexeme = "=", TokenType = Token_Class.AssignmentOp, Line = line, Column = col }); i++; col++; continue;
                    case '<': Tokens.Add(new Token { Id = tokenId++, Lexeme = "<", TokenType = Token_Class.LessThanOp, Line = line, Column = col }); i++; col++; continue;
                    case '>': Tokens.Add(new Token { Id = tokenId++, Lexeme = ">", TokenType = Token_Class.GreaterThanOp, Line = line, Column = col }); i++; col++; continue;
                    case '!': Tokens.Add(new Token { Id = tokenId++, Lexeme = "!", TokenType = Token_Class.NotOp, Line = line, Column = col }); i++; col++; continue;
                    case '(': Tokens.Add(new Token { Id = tokenId++, Lexeme = "(", TokenType = Token_Class.LParanthesis, Line = line, Column = col }); i++; col++; continue;
                    case ')': Tokens.Add(new Token { Id = tokenId++, Lexeme = ")", TokenType = Token_Class.RParanthesis, Line = line, Column = col }); i++; col++; continue;
                    case '{': Tokens.Add(new Token { Id = tokenId++, Lexeme = "{", TokenType = Token_Class.LBrace, Line = line, Column = col }); i++; col++; continue;
                    case '}': Tokens.Add(new Token { Id = tokenId++, Lexeme = "}", TokenType = Token_Class.RBrace, Line = line, Column = col }); i++; col++; continue;
                    case '[': Tokens.Add(new Token { Id = tokenId++, Lexeme = "[", TokenType = Token_Class.LBracket, Line = line, Column = col }); i++; col++; continue;
                    case ']': Tokens.Add(new Token { Id = tokenId++, Lexeme = "]", TokenType = Token_Class.RBracket, Line = line, Column = col }); i++; col++; continue;
                    case ':': Tokens.Add(new Token { Id = tokenId++, Lexeme = ":", TokenType = Token_Class.Colon, Line = line, Column = col }); i++; col++; continue;
                    case ';': Tokens.Add(new Token { Id = tokenId++, Lexeme = ";", TokenType = Token_Class.Semicolon, Line = line, Column = col }); i++; col++; continue;
                    case ',': Tokens.Add(new Token { Id = tokenId++, Lexeme = ",", TokenType = Token_Class.Comma, Line = line, Column = col }); i++; col++; continue;
                    case '.': Tokens.Add(new Token { Id = tokenId++, Lexeme = ".", TokenType = Token_Class.Dot, Line = line, Column = col }); i++; col++; continue;
                }

                // Numbers: IntConstant and FloatConstant
                if (char.IsDigit(ch))
                {
                    int sCol = col;
                    string num = "";
                    bool isFloat = false;
                    while (i < length && (char.IsDigit(sourceCode[i]) || sourceCode[i] == '.'))
                    {
                        if (sourceCode[i] == '.')
                        {
                            if (i + 1 < length && sourceCode[i + 1] == '.') break; // range ..
                            isFloat = true;
                        }
                        num += sourceCode[i];
                        i++; col++;
                    }

                    // Check for invalid trailing identifier chars: 123abc
                    string trailing = "";
                    while (i < length && (char.IsLetter(sourceCode[i]) || sourceCode[i] == '_'))
                    {
                        trailing += sourceCode[i];
                        i++; col++;
                    }

                    if (trailing.Length > 0)
                    {
                        Errors.Add(new CompilerError { Line = line, Column = sCol, Lexeme = num + trailing, Message = "Identifiers cannot begin with digits" });
                        Tokens.Add(new Token { Id = tokenId++, Lexeme = num + trailing, TokenType = Token_Class.Unknown, Line = line, Column = sCol });
                    }
                    else
                    {
                        Tokens.Add(new Token { Id = tokenId++, Lexeme = num, TokenType = isFloat ? Token_Class.FloatConstant : Token_Class.IntConstant, Line = line, Column = sCol });
                    }
                    continue;
                }

                // Identifiers & Keywords
                if (char.IsLetter(ch) || ch == '_')
                {
                    int sCol = col;
                    string ident = "";
                    while (i < length && (char.IsLetterOrDigit(sourceCode[i]) || sourceCode[i] == '_'))
                    {
                        ident += sourceCode[i];
                        i++; col++;
                    }

                    if (Keywords.ContainsKey(ident))
                        Tokens.Add(new Token { Id = tokenId++, Lexeme = ident, TokenType = Keywords[ident], Line = line, Column = sCol });
                    else
                        Tokens.Add(new Token { Id = tokenId++, Lexeme = ident, TokenType = Token_Class.Identifier, Line = line, Column = sCol });
                    continue;
                }

                // Unrecognized character
                Errors.Add(new CompilerError { Line = line, Column = col, Lexeme = ch.ToString(), Message = "Unrecognized character: " + ch });
                Tokens.Add(new Token { Id = tokenId++, Lexeme = ch.ToString(), TokenType = Token_Class.Unknown, Line = line, Column = col });
                i++; col++;
            }
        }
    }
